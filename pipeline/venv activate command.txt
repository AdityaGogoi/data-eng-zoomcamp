source .venv/bin/activate

-- Docker Postgresql command (IMPORTANT if you are trying to access Postgres DB):
docker run -it \
  -e POSTGRES_USER="root" \
  -e POSTGRES_PASSWORD="root" \
  -e POSTGRES_DB="ny_taxi" \
  -v $(pwd)/ny_taxi_postgres_data:/var/lib/postgresql \
  -p 5432:5432 \
  postgres:18

-- access Postgresql via command line:
uv run pgcli -h localhost -p 5432 -u root -d ny_taxi


-- Running ingestion script with params
uv run python ingest_data.py \
  --pg-user=root \
  --pg-pass=root \
  --pg-host=localhost \
  --pg-port=5432 \
  --pg-db=ny_taxi \
  --target-table=yellow_taxi_trips \
  --year=2021 \
  --month=1 \
  --chunksize=100000

-- Build Docker image for taxi ingestion
docker build -t taxi_ingest:v0001 .

-- Create docker network to connect DB and Ingest containers
docker network create pg-network

-- To list out all Docker networks
docker network ls

-- Run Docker Postgresql in Specified network:
docker run -it \
  -e POSTGRES_USER="root" \
  -e POSTGRES_PASSWORD="root" \
  -e POSTGRES_DB="ny_taxi" \
  -v $(pwd)/ny_taxi_postgres_data:/var/lib/postgresql \
  -p 5432:5432 \
  --network=pg-network \
  --name pgdatabase \
  postgres:18

-- Specialised UI for access this DB in a Specific network
-- Go to "PORTS" tab next to "TERMINAL", copy the "Forward Address" link next to 8085 Port.
-- Paste this link in new Browser tab.
-- For login, use PGADMIN_DEFAULT_EMAIL and PGADMIN_DEFAULT_PASSWORD values.
docker run -it \
  -e PGADMIN_DEFAULT_EMAIL="admin@admin.com" \
  -e PGADMIN_DEFAULT_PASSWORD="root" \
  -v pgadmin_data:/var/lib/pgadmin \
  -p 8085:80 \
  --network=pg-network \
  --name pgadmin \
  dpage/pgadmin4

-- Running ingestion script via Docker, accessing DB in another Docker image
docker run -it --rm\
  --network=pg-network \
  taxi_ingest:v0001 \
    --pg-user=root \
    --pg-pass=root \
    --pg-db=ny_taxi \
    --target-table=yellow_taxi_trips \
    --year=2021 \
    --month=1 \
    --chunksize=100000 \
    --pg-host=pgdatabase \
    --pg-port=5432


-- Run both DB and Admin container via docker-compose
-- Docker Netwrok created is "pipeline_default"
docker-compose up


docker run -it --rm\
  --network=pipeline_default \
  taxi_ingest:v0001 \
    --pg-user=root \
    --pg-pass=root \
    --pg-db=ny_taxi \
    --target-table=yellow_taxi_trips \
    --year=2021 \
    --month=1 \
    --chunksize=100000 \
    --pg-host=pgdatabase \
    --pg-port=5432
